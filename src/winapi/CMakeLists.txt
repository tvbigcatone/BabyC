cmake_minimum_required(VERSION 3.10)
project(winapi C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(WINAPI_THREADS "Eventi, thread, mutex, WaitFor*"       OFF)
option(WINAPI_SHMEM   "CreateFileMapping / MapViewOfFile"     OFF)
option(WINAPI_DISPLAY "EnumDisplayDevices (richiede X11)"     OFF)
option(WINAPI_UDEV    "GetDriveType con libudev"              OFF)
option(WINAPI_X11     "MessageBox grafica con X11"            OFF)

file(GLOB WINAPI_SOURCES CONFIGURE_DEPENDS src/*.c)

add_library(winapi STATIC ${WINAPI_SOURCES})
target_include_directories(winapi PUBLIC include PRIVATE src)
target_compile_options(winapi PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(winapi PUBLIC pthread rt m)

if(WINAPI_THREADS)
  target_compile_definitions(winapi PUBLIC ENABLE_WTHREADS=1)
endif()
if(WINAPI_SHMEM)
  target_compile_definitions(winapi PUBLIC ENABLE_WSHMEM=1)
endif()
if(WINAPI_DISPLAY)
  target_compile_definitions(winapi PUBLIC ENABLE_DISPLAY=1)
  target_link_libraries(winapi PUBLIC X11)
endif()
if(WINAPI_UDEV)
  target_compile_definitions(winapi PUBLIC ENABLE_UDEV=1)
  target_link_libraries(winapi PUBLIC udev)
endif()
if(WINAPI_X11)
  target_compile_definitions(winapi PUBLIC ENABLE_X11=1)
  target_link_libraries(winapi PUBLIC X11)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(winapi_smoke tests/smoke.c)
  target_link_libraries(winapi_smoke PRIVATE winapi)
  if(WINAPI_SHMEM)
    # Compile fault injection only into test-enabled shared-memory builds.
    target_compile_definitions(winapi PRIVATE WINAPI_TEST_FAULT_INJECTION=1)
    target_compile_definitions(winapi_smoke PRIVATE WINAPI_TEST_FAULT_INJECTION=1)
  endif()
  add_test(NAME winapi_smoke COMMAND winapi_smoke)
endif()
